import processing.serial.*;

import processing.sound.*;

SoundFile BassDrum;
SoundFile SnareDrum;
SoundFile HiHat;
SoundFile ClosedHiHat;

float[] sensorValues = {0,0,0,0};
String rawSerial = "";
float masterVolume = 0.8;
int volumeStep = 2;
ArrayList<Burst> bursts = new ArrayList<Burst>();


boolean [] drumFlags = {true, true, true, true};

Serial connection;

void setup(){
  size(800, 800);
  
  BassDrum = new SoundFile(this, "BT0AADA.WAV");
  SnareDrum = new SoundFile(this, "CLOP4.WAV");
  HiHat = new SoundFile(this, "HHCDA.WAV");
  ClosedHiHat = new SoundFile(this, "LT7DA.WAV");
  printArray(Serial.list());
  connection = new Serial(this, Serial.list()[2], 115200);
  connection.bufferUntil('\n');
  updateVolume();
  bursts.clear();
}

void draw(){
  background(0);
  
  for(int i = bursts.size() - 1; i >= 0; i--){
    Burst b = bursts.get(i);
    b.display();
    if(b.isDead()){
      bursts.remove(i);
    }
  }
}

void serialEvent(Serial conn){
  String incoming = conn.readString();
  String [] values = split(trim(incoming),',');
  printArray(values);
  
  rawSerial = incoming;
  if(values.length == 4){
  sensorValues[0] = float(values[0]);
  sensorValues[1] = float(values[1]);
  sensorValues[2] = float(values[2]);
  sensorValues[3] = float(values[3]);
}
  
  if(values. length == 4){
    if(float(values[0]) > 0){
      if(!BassDrum.isPlaying()&&drumFlags[0]){
        BassDrum.play();
        spawnBurst();
        drumFlags[0] = false;
      }
    }
    else {
      drumFlags[0] = true;
    }
    
    if(float(values[1]) > 0){
      if(!SnareDrum.isPlaying()&&drumFlags[1]){
        SnareDrum.play();
        spawnBurst();
        drumFlags[1] = false;
      }
    }
    else {
      drumFlags[1] = true;
    }
    
    if(float(values[2]) > 0){
      if(!HiHat.isPlaying()&&drumFlags[2]){
        HiHat.play();
        spawnBurst();
        drumFlags[2] = false;
      }
    }
    else {
      drumFlags[2] = true;
    }
    
    if(float(values[3]) > 0){
      if(!ClosedHiHat.isPlaying()&&drumFlags[3]){
        ClosedHiHat.play();
        spawnBurst();
        
        volumeStep = (volumeStep + 1) % 4;
        if(volumeStep == 0) masterVolume = 0.25;
        if(volumeStep == 1) masterVolume = 0.5;
        if(volumeStep == 2) masterVolume = 0.75;
        if(volumeStep == 3) masterVolume = 1.0;
        updateVolume();

        drumFlags[3] = false;
      }
    }
    else {
      drumFlags[3] = true;
    }
  }
}

void updateVolume(){
  BassDrum.amp(masterVolume);
  SnareDrum.amp(masterVolume);
  HiHat.amp(masterVolume);
  ClosedHiHat.amp(masterVolume);
}

void spawnBurst(){
  bursts.add(new Burst(random(width), random(height), random(20, 120), color(random(255), random(255), random(255))));
}

class Burst{
  float x;
  float y;
  float r;
  int c;
  int bornMs;
  
  Burst(float x, float y, float r, int c){
    this.x = x;
    this.y = y;
    this.r = r;
    this.c = c;
    bornMs = millis();
  }
  
  void display(){
    int age = millis() - bornMs;
    float a = map(age, 0, 1000, 255, 0);
    a = constrain(a, 0, 255);
    noStroke();
    fill(red(c), green(c), blue(c), a);
    ellipse(x, y, r, r);
  }
  
  boolean isDead(){
    return millis() - bornMs >= 1000;
  }
}
